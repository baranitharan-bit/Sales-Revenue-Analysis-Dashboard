# Helper function to parse uploaded files
def parse_contents(contents, filename):
    content_type, content_string = contents.split(',')
    decoded = base64.b64decode(content_string)
    try:
        if 'csv' in filename:
            df = pd.read_csv(io.StringIO(decoded.decode('utf-8')))
        elif 'xls' in filename:
            df = pd.read_excel(io.BytesIO(decoded))
        else:
            return None
        
        # Format columns if they exist
        if 'Date' in df.columns:
            df['Date'] = pd.to_datetime(df['Date'])
        return df
    except Exception as e:
        print(e)
        return None

# Callback to process data upload and set options
@app.callback(
    [Output('stored-data', 'data'),
     Output('upload-status', 'children'),
     Output('product-filter', 'options'),
     Output('region-filter', 'options')],
    [Input('upload-data', 'contents')],
    [State('upload-data', 'filename')]
)
def update_dataset(contents, filename):
    if contents is not None:
        df = parse_contents(contents, filename)
        if df is not None:
            status = f"Successfully loaded: {filename}"
        else:
            df = sample_data
            status = "Failed to process file. Showing default sample data."
    else:
        df = sample_data
        status = "Using default sample dataset. Upload your own CSV/Excel to analyze custom data."

    product_opts = [{'label': i, 'value': i} for i in df['Product'].unique()] if 'Product' in df.columns else []
    region_opts = [{'label': i, 'value': i} for i in df['Region'].unique()] if 'Region' in df.columns else []

    return df.to_dict('records'), status, product_opts, region_opts

# Callback to update Dashboard KPIs and Visualizations
@app.callback(
    [Output('kpi-total-revenue', 'children'),
     Output('kpi-total-sales', 'children'),
     Output('kpi-avg-order', 'children'),
     Output('revenue-trend-chart', 'figure'),
     Output('top-products-chart', 'figure'),
     Output('regional-revenue-chart', 'figure')],
    [Input('stored-data', 'data'),
     Input('product-filter', 'value'),
     Input('region-filter', 'value')]
)
def update_dashboard(data, selected_products, selected_regions):
    df = pd.DataFrame(data)

    # Filter by selected dropdown values
    if selected_products:
        df = df[df['Product'].isin(selected_products)]
    if selected_regions:
        df = df[df['Region'].isin(selected_regions)]

    # Compute KPIs
    total_rev = df['Revenue'].sum() if 'Revenue' in df.columns else 0
    total_sales = df['Sales_Quantity'].sum() if 'Sales_Quantity' in df.columns else 0
    avg_order = df['Revenue'].mean() if 'Revenue' in df.columns else 0

    kpi_rev = [html.H3("Total Revenue"), html.H2(f"${total_rev:,.2f}")]
    kpi_sales = [html.H3("Units Sold"), html.H2(f"{total_sales:,}")]
    kpi_avg = [html.H3("Avg Order Value"), html.H2(f"${avg_order:,.2f}")]

    # 1. Revenue Trend Line Chart
    if 'Date' in df.columns and 'Revenue' in df.columns:
        df_trend = df.groupby('Date')['Revenue'].sum().reset_index()
        fig_trend = px.line(df_trend, x='Date', y='Revenue', title="Revenue Trend Over Time", markers=True)
        fig_trend.update_traces(line_color='#1f77b4')
    else:
        fig_trend = px.line(title="Date/Revenue columns missing")

    # 2. Top Performing Products Bar Chart
    if 'Product' in df.columns and 'Revenue' in df.columns:
        df_prod = df.groupby('Product')['Revenue'].sum().reset_index().sort_values(by='Revenue', ascending=True)
        fig_prod = px.bar(df_prod, x='Revenue', y='Product', orientation='h', title="Revenue by Product", text_auto='.2s')
        fig_prod.update_traces(marker_color='#2ca02c')
    else:
        fig_prod = px.bar(title="Product/Revenue columns missing")

    # 3. Regional Distribution Donut Chart
    if 'Region' in df.columns and 'Revenue' in df.columns:
        df_region = df.groupby('Region')['Revenue'].sum().reset_index()
        fig_region = px.pie(df_region, values='Revenue', names='Region', title="Regional Revenue Breakdown", hole=0.4)
    else:
        fig_region = px.pie(title="Region/Revenue columns missing")

    return kpi_rev, kpi_sales, kpi_avg, fig_trend, fig_prod, fig_region
